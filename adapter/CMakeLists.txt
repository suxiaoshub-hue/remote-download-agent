cmake_minimum_required(VERSION 3.15)
project(PcstoryAdapter LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0601)
if(MSVC)
    add_compile_options(/W4 /utf-8)
endif()
add_executable(PcstoryStatusTests status.cpp tests/status.cpp)
add_executable(PcstoryCommandFixture tests/fixture.cpp)
target_link_libraries(PcstoryCommandFixture PRIVATE user32)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/main.cpp")
    add_executable(PcstoryAdapter main.cpp status.cpp)
    target_link_libraries(PcstoryAdapter PRIVATE user32 bcrypt)
    add_executable(PcstoryCommandTest main.cpp status.cpp)
    target_compile_definitions(PcstoryCommandTest PRIVATE PCSTORY_TEST_FIXTURE)
    target_link_libraries(PcstoryCommandTest PRIVATE user32 bcrypt)
endif()
